# API Backend — Lambda Dependencies
# Pinned to manylinux2014_aarch64 compatible versions for arm64 Lambda
# Build with: pip install --platform manylinux2014_aarch64 --only-binary=:all: -r requirements.txt

# AWS + AI
langchain-aws==0.2.11

# main.py imports langchain.chains.{retrieval,combine_documents}. This used to
# arrive transitively via langchain-community, which is now local-only — so it
# is pinned explicitly. Relying on a transitive for something we import directly
# is exactly what caused the greenlet outage; do not unpin it.
langchain==0.3.25

# SQLAlchemy (pulled in by langchain, above) needs greenlet for its asyncio
# support, but declares it behind an environment marker on platform_machine.
# build.sh cross-builds with --platform manylinux2014_aarch64 from an x86_64
# runner, and pip evaluates markers against the HOST, so the marker is false
# and greenlet is silently dropped. The Lambda then fails at import with
# "requires that the Python 'greenlet' library is installed". Pinned
# explicitly so the cross-build cannot omit it.
greenlet==3.2.5

# Vector store is Amazon S3 Vectors, reached through the boto3 client that the
# bundled botocore already provides — no database driver, no connection pool.
# langchain-postgres/psycopg moved to requirements-local.txt: pgvector is now
# only used by the local no-AWS path.

# Web framework
fastapi==0.111.0
mangum==0.17.0

# Rate limiting (application-level defense-in-depth layer)
slowapi==0.1.9

# Input validation
pydantic==2.7.4
